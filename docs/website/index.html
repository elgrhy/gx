hello.gx 
agent "greeter" {
  remember {
    name = "World"
  }

  when started {
    say "Hello, {memory.name}!"
  }
}
ai_agent.gx 
agent "assistant" {
  when started {
    result = ask openai {
      prompt: "Summarize: {memory.text}"
    }
    if result.confidence > 0.8 {
      say result.text
    } else {
      escalate to human
    }
  }
}
functions.gx 
function classify(text) {
  label = infer classifier {
    input: text,
    classes: ["support", "sales", "spam"]
  }
  return label
}

agent "router" {
  when started {
    kind = classify(memory.message)
    say "Routed to: {kind}"
  }
}
packages.gx 
use js.axios
use py.requests

agent "fetcher" {
  when started {
    try {
      data = js.axios.get(memory.url)
      say data.status
    } catch err {
      log("Error: " + err)
    }
  }
}
Why GX

Built for the AI era

🔍

Fully auditable

Every AI call is auto-logged to memory.ai_trace. Confidence scores built in. No black boxes.

🧠

Brain cycle architecture

Every agent follows Plan → Execute → Remember → Communicate. Clear, predictable, debuggable.

🤖

AI primitives in the language

ask openai, ask anthropic, ask ollama, embed, infer classifier — first-class syntax.

📦

Zero ecosystem lock-in

Use any npm, pip, or cargo package directly. use js.axios, use py.pandas — it just works.

⚡

Rust performance

Tree-walking interpreter written in Rust. Fast startup, no GC pauses, runs on any platform.

🛡️

Safe escalation

When confidence is low, escalate to human stops the agent and signals for review — by design.
